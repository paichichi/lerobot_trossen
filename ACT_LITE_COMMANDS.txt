# Backbone-ACT-lite: ours_rn50 upstream backbone + official LeRobot ACT/Trossen
# Run from the lerobot_trossen repository root.
set -euo pipefail

# ============================================================================
# COPY/PASTE: download one uploaded checkpoint and prepare real-robot eval
# ============================================================================
# Default: our frozen ours_rn50 backbone + ACT-lite head.
# To use official ACT ResNet18 instead, change this one line to: ACT_MODEL=rn18
ACT_MODEL=ours_rn50

case "$ACT_MODEL" in
  ours_rn50)
    POLICY_REPO=Chipaipai/act-lite-ours-rn50-carrot-100
    POLICY_REVISION=424a8456d2e43f3730e5d684b122ed151728ec23
    ;;
  rn18)
    POLICY_REPO=Chipaipai/act-official-rn18-carrot-100
    POLICY_REVISION=8f3cf3b8358d46928bc12271027787cc1f7b0499
    ;;
  *)
    echo "ACT_MODEL must be ours_rn50 or rn18" >&2
    exit 2
    ;;
esac

# Python 3.12 is required by the locked LeRobot 0.6 environment.
uv sync --extra act
POLICY_PATH="$PWD/checkpoints/$ACT_MODEL"
uv run --no-sync hf download "$POLICY_REPO" \
  --revision "$POLICY_REVISION" \
  --local-dir "$POLICY_PATH"

# ours_rn50 needs the upstream checkpoint and TCC source tree while constructing
# the model. The uploaded ACT model then restores the complete trained state.
# Change BACKBONE_SOURCE_ROOT only if TCC-core lives elsewhere on this computer.
if [[ "$ACT_MODEL" == ours_rn50 ]]; then
  export BACKBONE_SOURCE_ROOT="${BACKBONE_SOURCE_ROOT:-/home/robotarm/TCC-core}"
  export BACKBONE_CHECKPOINT="$PWD/assets/tcc-policy-assets/backbones/ours_rn50/checkpoint_040000.pt"
  uv run --no-sync hf download Chipaipai/tcc-core-real-robot-policies \
    backbones/ours_rn50/checkpoint_040000.pt \
    --revision 466bc1e4be7d1899da7281a5d8a30add04bf7e3c \
    --local-dir="$PWD/assets/tcc-policy-assets"
  test -f "$BACKBONE_SOURCE_ROOT/xirl/models.py" || {
    echo "TCC source missing: $BACKBONE_SOURCE_ROOT/xirl/models.py" >&2
    exit 1
  }
fi

# Read this before connecting the robot. Both uploaded checkpoints are currently
# marked BLOCKED by the conservative offline first-frame gate. The official
# Trossen driver still clips every arm target to measured position +/-0.07 rad
# and the gripper target to +/-0.003 m, but this is not a success guarantee.
uv run --no-sync python -m json.tool "$POLICY_PATH/first_frame_report.json"

# PHYSICAL ROBOT COMMAND — paste only with the arm supervised and E-stop ready.
# The arm first moves to official Trossen staged_positions. Control runs at
# 20 Hz and min_time_to_move=2/20=0.1 s. Stop with Ctrl-C if motion is wrong.
uv run --no-sync lerobot-record \
  --robot.discover_packages_path=lerobot_robot_trossen \
  --robot.type=widowxai_follower_robot \
  --robot.ip_address=192.168.1.4 \
  --robot.id=follower \
  --robot.loop_rate=20 \
  --robot.min_time_to_move_multiplier=2.0 \
  --robot.max_relative_target='{"joint_0": 0.07, "joint_1": 0.07, "joint_2": 0.07, "joint_3": 0.07, "joint_4": 0.07, "joint_5": 0.07, "left_carriage_joint": 0.003}' \
  --robot.cameras='{cam_main: {type: intelrealsense, serial_number_or_name: "838212073584", width: 640, height: 480, fps: 30}, cam_wrist: {type: intelrealsense, serial_number_or_name: "409122274608", width: 640, height: 480, fps: 30}}' \
  --dataset.repo_id="Chipaipai/act-${ACT_MODEL}-carrot-eval" \
  --dataset.num_episodes=1 \
  --dataset.episode_time_s=30 \
  --dataset.reset_time_s=10 \
  --dataset.single_task="Pick up the carrot and place it in the pan" \
  --dataset.push_to_hub=false \
  --display_data=true \
  --policy.discover_packages_path=lerobot_policy_backbone_act \
  --policy.path="$POLICY_PATH"

# ============================================================================
# TRAINING RECIPE (not needed when using the uploaded checkpoints above)
# ============================================================================

# 1. Install only the policy and exact locked LeRobot training runtime. This
#    avoids pulling hardware/teleoperator build dependencies onto the GPU host.
uv sync --package lerobot-policy-backbone-act --extra train --group dev

# TorchCodec removes PyAV's random-seek bottleneck. This helper only extracts
# the missing Ubuntu FFmpeg 6 runtime libraries under .local/; it does not use
# sudo or modify the system installation.
bash scripts/bootstrap_torchcodec_ffmpeg6.sh
export ACT_FFMPEG_LIB_DIR="$PWD/.local/ffmpeg6/usr/lib/x86_64-linux-gnu"
export LD_LIBRARY_PATH="$ACT_FFMPEG_LIB_DIR${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"

# LeRobot 0.6 delegates mixed precision to Accelerate. RTX 5090 supports BF16
# natively; model parameters and saved checkpoints remain compatible with the
# standard LeRobot loading path.
export ACCELERATE_MIXED_PRECISION=bf16

# 2. Point this at the carrot dataset folder containing meta/, data/, videos/.
export ACT_DATASET_ROOT="$PWD/datasets/pick_and_place_4_object_diverse/pick_and_place_carrot_100"

# 3. Validate the original LeRobot v2.1 dataset.
uv run --no-sync python scripts/check_act_dataset.py "$ACT_DATASET_ROOT"

# 4. One-time IN-PLACE conversion to LeRobot v3.0. Make a backup first.
#    This stays local and does not modify the Hugging Face dataset.
uv run --no-sync python -m lerobot.scripts.convert_dataset_v21_to_v30 \
  --repo-id=UoA-Trossen-Arm/pick_and_place_carrot_100 \
  --root="$ACT_DATASET_ROOT" \
  --push-to-hub=false

# 5. Confirm the converted dataset contract.
uv run --no-sync python scripts/check_act_dataset.py \
  "$ACT_DATASET_ROOT" \
  --require-v3

# 6. Download the upstream ours_rn50 backbone used by the experiment.
hf download Chipaipai/tcc-core-real-robot-policies \
  backbones/ours_rn50/checkpoint_040000.pt \
  --local-dir=assets/tcc-policy-assets

export BACKBONE_SOURCE_ROOT="${BACKBONE_SOURCE_ROOT:-/home/paichichi/projects/TCC-core}"
export BACKBONE_CHECKPOINT="$PWD/assets/tcc-policy-assets/backbones/ours_rn50/checkpoint_040000.pt"
export ACT_OUTPUT_DIR="$PWD/outputs/train/backbone_act_lite_ours_rn50_carrot_100"
export ACT_TRAIN_LOG="$ACT_OUTPUT_DIR.train.log"

# Match original ACT: randomly shuffle complete episodes with a fixed seed, then
# hold out 20% per task for validation. There is no offline test-demonstration set.
export ACT_EPISODE_ORDER="$(uv run --no-sync python scripts/make_act_episode_order.py \
  --repo-id=UoA-Trossen-Arm/pick_and_place_carrot_100 \
  --root="$ACT_DATASET_ROOT" \
  --seed=1000)"
mkdir -p "$(dirname "$ACT_OUTPUT_DIR")"

# 7. Train lightweight ACT. Two image keys are discovered from dataset metadata.
#    At 20 Hz: chunk_size=40 predicts 2 s, n_action_steps=10 executes 0.5 s
#    before replanning. The upstream backbone is frozen in the first controlled
#    experiment, so performance still measures the learned representation.
uv run --no-sync lerobot-train \
  --dataset.repo_id=UoA-Trossen-Arm/pick_and_place_carrot_100 \
  --dataset.root="$ACT_DATASET_ROOT" \
  --dataset.episodes="$ACT_EPISODE_ORDER" \
  --dataset.eval_split=0.2 \
  --dataset.return_uint8=true \
  --dataset.video_backend=torchcodec \
  --policy.discover_packages_path=lerobot_policy_backbone_act \
  --policy.type=backbone_act \
  --policy.repo_id=Chipaipai/act-lite-ours-rn50-carrot-100 \
  --policy.push_to_hub=false \
  --policy.device=cuda \
  --policy.backbone_checkpoint="$BACKBONE_CHECKPOINT" \
  --policy.backbone_source_root="$BACKBONE_SOURCE_ROOT" \
  --policy.backbone_family=ours_rn50 \
  --policy.freeze_vision_backbone=true \
  --policy.chunk_size=40 \
  --policy.n_action_steps=10 \
  --policy.vision_backbone=resnet50 \
  --policy.dim_model=256 \
  --policy.n_heads=4 \
  --policy.dim_feedforward=1024 \
  --policy.n_encoder_layers=2 \
  --policy.n_decoder_layers=1 \
  --policy.n_vae_encoder_layers=2 \
  --batch_size=16 \
  --num_workers=8 \
  --prefetch_factor=4 \
  --steps=50000 \
  --eval_steps=2000 \
  --save_freq=2000 \
  --log_freq=100 \
  --output_dir="$ACT_OUTPUT_DIR" \
  --job_name=backbone_act_lite_ours_rn50_carrot_100 \
  --wandb.enable=false \
  2>&1 | tee "$ACT_TRAIN_LOG"

# Original ACT deploys the checkpoint with the lowest validation loss. Because
# eval_steps and save_freq match, every evaluated candidate exists on disk.
uv run --no-sync python scripts/select_best_act_checkpoint.py \
  "$ACT_OUTPUT_DIR" \
  --log-path="$ACT_TRAIN_LOG"

# 8. Real-robot evaluation. Replace POLICY_PATH with a local checkpoint or HF repo.
#    LeRobot connects the arm and moves to Trossen staged_positions (the same
#    pose as dataset_collection_home). Cameras stream at the collection setting
#    of 30 FPS; the robot/control and dataset sampling rate remain 20 Hz.
export POLICY_PATH="$(< "$ACT_OUTPUT_DIR/best_checkpoint.txt")"

# On the robot-control computer, install the complete hardware workspace once.
# Ubuntu requires a C compiler and Python headers for the evdev dependency, e.g.
# `sudo apt install build-essential python3.12-dev`, before this sync.
uv sync --extra act

uv run --no-sync lerobot-record \
  --robot.discover_packages_path=lerobot_robot_trossen \
  --robot.type=widowxai_follower_robot \
  --robot.ip_address=192.168.1.4 \
  --robot.id=follower \
  --robot.loop_rate=20 \
  --robot.min_time_to_move_multiplier=2.0 \
  --robot.max_relative_target='{"joint_0": 0.07, "joint_1": 0.07, "joint_2": 0.07, "joint_3": 0.07, "joint_4": 0.07, "joint_5": 0.07, "left_carriage_joint": 0.003}' \
  --robot.cameras='{cam_main: {type: intelrealsense, serial_number_or_name: "838212073584", width: 640, height: 480, fps: 30}, cam_wrist: {type: intelrealsense, serial_number_or_name: "409122274608", width: 640, height: 480, fps: 30}}' \
  --dataset.repo_id=Chipaipai/act-lite-carrot-eval \
  --dataset.num_episodes=5 \
  --dataset.episode_time_s=30 \
  --dataset.reset_time_s=10 \
  --dataset.single_task="Pick up the carrot and place it in the pan" \
  --dataset.push_to_hub=false \
  --display_data=true \
  --policy.discover_packages_path=lerobot_policy_backbone_act \
  --policy.path="$POLICY_PATH"
