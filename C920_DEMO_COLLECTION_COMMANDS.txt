# C920 单摄像头 Demo 练习与正式采集
#
# 硬件：
#   follower: 192.168.1.4
#   leader:   192.168.1.2
#   camera:   Logitech C920（仅 cam_main）
#
# 安全：开始前清空机械臂工作区，并保持 E-stop 随时可按。
# 不要同时运行 rollout、viewer 和正式录制；C920 同一时间只能由一个程序占用。


==============================
1. 练习模式（不保存数据）
==============================

# 先确认没有 rollout 或录制程序占用 C920：
fuser -v /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0

# 终端 1：锁定 C920 对焦并打开低延迟画面。
# 按 q 关闭 viewer。
uv run --no-sync python scripts/lock_c920_focus.py \
  --device /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0

ffplay -fflags nobuffer -flags low_delay -framedrop \
  -f v4l2 \
  -input_format mjpeg \
  -video_size 640x480 \
  -framerate 20 \
  /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0

# 终端 2：进入 leader → follower 练习模式，不保存任何数据。
# 操作结束后按 Ctrl+C。
uv run lerobot-teleoperate \
  --robot.discover_packages_path=lerobot_robot_trossen \
  --robot.type=widowxai_follower_robot \
  --robot.ip_address=192.168.1.4 \
  --robot.id=follower \
  --robot.loop_rate=20 \
  --robot.min_time_to_move_multiplier=1.5 \
  --robot.max_relative_target=0.07 \
  --teleop.type=widowxai_leader_teleop \
  --teleop.ip_address=192.168.1.2 \
  --teleop.id=leader \
  --display_data=false


==============================
2. 正式采集前准备
==============================

# 先关闭上面的 ffplay（按 q）和练习模式（Ctrl+C）。
# 再检查 C920 已释放；无 PID 输出才可以继续。
fuser -v /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0

# 每录一条只修改 DEMO_ID，例如 001、002、003。
# 每条 demo 使用独立目录，失败的 demo 可以单独处理，不影响其他数据。
DEMO_ID=001
DATASET_NAME="carrot_to_pot_demo_${DEMO_ID}"
DATASET_ROOT="datasets/carrot_to_pot_raw/${DATASET_NAME}"

# 防止覆盖已有 demo。看到 already exists 时，换一个 DEMO_ID。
if [[ -e "$DATASET_ROOT" ]]; then
  echo "Demo already exists: $DATASET_ROOT"
  false
fi

# 锁定 C920 的对焦、曝光和白平衡，然后开始录制。
uv run --no-sync python scripts/lock_c920_focus.py \
  --device /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0


==============================
3. 正式采集：每次只录 1 条 Demo
==============================

uv run lerobot-record \
  --robot.discover_packages_path=lerobot_robot_trossen \
  --robot.type=widowxai_follower_robot \
  --robot.ip_address=192.168.1.4 \
  --robot.id=follower \
  --robot.loop_rate=20 \
  --robot.min_time_to_move_multiplier=1.5 \
  --robot.max_relative_target=0.07 \
  --robot.cameras='{cam_main: {type: opencv, index_or_path: /dev/v4l/by-id/usb-046d_HD_Pro_Webcam_C920-video-index0, width: 640, height: 480, fps: 20, fourcc: MJPG}}' \
  --teleop.type=widowxai_leader_teleop \
  --teleop.ip_address=192.168.1.2 \
  --teleop.id=leader \
  --display_data=false \
  --dataset.repo_id="local/${DATASET_NAME}" \
  --dataset.root="$DATASET_ROOT" \
  --dataset.fps=20 \
  --dataset.episode_time_s=15 \
  --dataset.num_episodes=1 \
  --dataset.video=true \
  --dataset.push_to_hub=false \
  --dataset.single_task="Pick up the carrot and place it in the pot." \
  --play_sounds=true


==============================
4. 录下一条
==============================

# 确认上一条成功并退出后，重新放置胡萝卜和 pot。
# 然后修改 ID，并从“正式采集前准备”重新执行：
DEMO_ID=002

# 不要只修改 repo_id 而复用旧 root；repo_id 和 DATASET_ROOT 必须随 ID 一起变化。
# 录制失败时不要覆盖原目录，先保留现场并记录失败 ID，之后再决定是否移入 backup。


==============================
5. 一条好 Demo 的快速标准
==============================

# 1) 开始帧中胡萝卜、pot、夹爪都清晰可见。
# 2) 动作连续、目标明确，不在空中反复犹豫或来回修正。
# 3) 夹爪在胡萝卜处一次闭合，搬运过程中保持夹紧。
# 4) 胡萝卜完整进入 pot 后再松开。
# 5) 15 秒内自然完成；完成后短暂停住，不做多余动作。
# 6) 没有碰撞、砸桌、0.07 安全截断警告或相机失焦。
